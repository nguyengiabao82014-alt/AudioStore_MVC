using System.Web.Mvc;
using System.Web.Routing;

namespace AudioStore.Areas.Admin.Controllers
{
    // Kế thừa từ Controller mặc định
    public class BaseAdminController : Controller
    {
        // Hàm này sẽ tự động kích hoạt TRƯỚC KHI bất kỳ trang Admin nào được tải
        protected override void OnActionExecuting(ActionExecutingContext filterContext)
        {
            // Kiểm tra: Nếu chưa đăng nhập HOẶC RoleID không phải là 1 (không phải Admin)
            if (Session["User"] == null || Session["RoleID"] == null || (int)Session["RoleID"] != 1)
            {
                // Đá văng người dùng về trang Đăng nhập ở khu vực người dùng (area = "")
                filterContext.Result = new RedirectToRouteResult(
                    new RouteValueDictionary(
                        new { controller = "Account", action = "Login", area = "" }
                    )
                );
            }

            // Nếu đúng là Admin, cho phép đi tiếp
            base.OnActionExecuting(filterContext);
        }
    }
}

using System;
using System.Linq;
using System.Web.Mvc;
using AudioStore.Models;

namespace AudioStore.Areas.Admin.Controllers
{
    public class DiscountCodesController : Controller
    {
        private DB_AudioStoreEntities db = new DB_AudioStoreEntities();

        private bool CheckAdmin()
        {
            if (Session["User"] is Users u && u.RoleID == 1) return true;
            return false;
        }

        public ActionResult Index()
        {
            if (!CheckAdmin()) return RedirectToAction("Login", "Account", new { area = "" });
            // Sắp xếp theo Code thay vì DiscountID
            var list = db.DiscountCodes.OrderByDescending(d => d.Code).ToList();
            return View(list);
        }

        [HttpGet]
        public ActionResult Create()
        {
            if (!CheckAdmin()) return RedirectToAction("Login", "Account", new { area = "" });
            return View();
        }

        [HttpPost]
        [ValidateAntiForgeryToken]
        public ActionResult Create(DiscountCodes model)
        {
            if (!CheckAdmin()) return RedirectToAction("Login", "Account", new { area = "" });

            if (ModelState.IsValid)
            {
                var existing = db.DiscountCodes.FirstOrDefault(d => d.Code == model.Code);
                if (existing != null)
                {
                    ModelState.AddModelError("Code", "Mã giảm giá này đã tồn tại trong hệ thống!");
                    return View(model);
                }

                model.IsActive = true;
                db.DiscountCodes.Add(model);
                db.SaveChanges();
                return RedirectToAction("Index");
            }
            return View(model);
        }

        // Đổi int id thành string id vì khóa chính Code là kiểu chuỗi
        public ActionResult Delete(string id)
        {
            if (!CheckAdmin()) return RedirectToAction("Login", "Account", new { area = "" });
            var item = db.DiscountCodes.Find(id);
            if (item != null)
            {
                db.DiscountCodes.Remove(item);
                db.SaveChanges();
            }
            return RedirectToAction("Index");
        }
    }
}

using System;
using System.Collections.Generic;
using System.Data;
using System.Data.Entity;
using System.IO; // Thư viện xử lý File & Directory
using System.Linq;
using System.Net;
using System.Web;
using System.Web.Mvc;
using AudioStore.Models;

namespace AudioStore.Areas.Admin.Controllers
{
    public class ProductsController : BaseAdminController
    {
        private DB_AudioStoreEntities db = new DB_AudioStoreEntities();

        // GET: Admin/Products
        public ActionResult Index()
        {
            var products = db.Products.Include(p => p.Brands).Include(p => p.Categories);
            return View(products.ToList());
        }

        // GET: Admin/Products/Details/5
        public ActionResult Details(int? id)
        {
            if (id == null)
            {
                return new HttpStatusCodeResult(HttpStatusCode.BadRequest);
            }
            Products products = db.Products.Find(id);
            if (products == null)
            {
                return HttpNotFound();
            }
            return View(products);
        }

        // GET: Admin/Products/Create
        public ActionResult Create()
        {
            ViewBag.BrandID = new SelectList(db.Brands, "BrandID", "BrandName");
            ViewBag.CategoryID = new SelectList(db.Categories, "CategoryID", "CategoryName");
            return View();
        }

        // POST: Admin/Products/Create
        [HttpPost]
        [ValidateAntiForgeryToken]
        public ActionResult Create([Bind(Include = "ProductID,ProductName,Price,StockQuantity,CategoryID,BrandID,IsWireless,ImageUrl,VideoUrl,Specifications,Description,IsActive")] Products products, IEnumerable<HttpPostedFileBase> imageFiles)
        {
            if (ModelState.IsValid)
            {
                // --- XỬ LÝ UPLOAD NHIỀU ẢNH AN TOÀN VỚI SaveAs ---
                if (imageFiles != null && imageFiles.Any(f => f != null && f.ContentLength > 0))
                {
                    List<string> imagePaths = new List<string>();

                    string folderPath = Server.MapPath("~/Content/Images/products/");
                    if (!Directory.Exists(folderPath))
                    {
                        Directory.CreateDirectory(folderPath);
                    }

                    foreach (var file in imageFiles)
                    {
                        if (file != null && file.ContentLength > 0)
                        {
                            // Lấy phần mở rộng thực tế của file (.jpg, .png, .webp...)
                            string extension = Path.GetExtension(file.FileName);
                            if (string.IsNullOrEmpty(extension)) extension = ".jpg";

                            // Tạo tên file duy nhất tránh trùng đè
                            string fileName = "product-" + Guid.NewGuid().ToString().Substring(0, 8) + extension;
                            string path = Path.Combine(folderPath, fileName);

                            // Ghi file trực tiếp xuống ổ cứng (không thông qua GDI+ gây lỗi Parameter is not valid)
                            file.SaveAs(path);

                            // Thêm đường dẫn tương đối vào danh sách
                            imagePaths.Add("/Content/Images/products/" + fileName);
                        }
                    }

                    // Nối danh sách các đường dẫn lại với nhau bằng dấu chấm phẩy (;)
                    products.ImageUrl = string.Join(";", imagePaths);
                }

                db.Products.Add(products);
                db.SaveChanges();
                return RedirectToAction("Index");
            }

            ViewBag.BrandID = new SelectList(db.Brands, "BrandID", "BrandName", products.BrandID);
            ViewBag.CategoryID = new SelectList(db.Categories, "CategoryID", "CategoryName", products.CategoryID);
            return View(products);
        }

        // GET: Admin/Products/Edit/5
        public ActionResult Edit(int? id)
        {
            if (id == null)
            {
                return new HttpStatusCodeResult(HttpStatusCode.BadRequest);
            }
            Products products = db.Products.Find(id);
            if (products == null)
            {
                return HttpNotFound();
            }
            ViewBag.BrandID = new SelectList(db.Brands, "BrandID", "BrandName", products.BrandID);
            ViewBag.CategoryID = new SelectList(db.Categories, "CategoryID", "CategoryName", products.CategoryID);
            return View(products);
        }

        // POST: Admin/Products/Edit/5
        [HttpPost]
        [ValidateAntiForgeryToken]
        public ActionResult Edit([Bind(Include = "ProductID,ProductName,Price,StockQuantity,CategoryID,BrandID,IsWireless,ImageUrl,VideoUrl,Specifications,Description,IsActive")] Products products, IEnumerable<HttpPostedFileBase> imageFiles)
        {
            if (ModelState.IsValid)
            {
                // --- XỬ LÝ CẬP NHẬT NHIỀU ẢNH KHI EDIT ---
                if (imageFiles != null && imageFiles.Any(f => f != null && f.ContentLength > 0))
                {
                    List<string> imagePaths = new List<string>();

                    string folderPath = Server.MapPath("~/Content/Images/products/");
                    if (!Directory.Exists(folderPath))
                    {
                        Directory.CreateDirectory(folderPath);
                    }

                    foreach (var file in imageFiles)
                    {
                        if (file != null && file.ContentLength > 0)
                        {
                            string extension = Path.GetExtension(file.FileName);
                            if (string.IsNullOrEmpty(extension)) extension = ".jpg";

                            string fileName = "product-" + Guid.NewGuid().ToString().Substring(0, 8) + extension;
                            string path = Path.Combine(folderPath, fileName);

                            file.SaveAs(path);

                            imagePaths.Add("/Content/Images/products/" + fileName);
                        }
                    }

                    products.ImageUrl = string.Join(";", imagePaths);
                }

                db.SaveChanges();
                return RedirectToAction("Index");
            }
            ViewBag.BrandID = new SelectList(db.Brands, "BrandID", "BrandName", products.BrandID);
            ViewBag.CategoryID = new SelectList(db.Categories, "CategoryID", "CategoryName", products.CategoryID);

using System;
using System.Linq;
using System.Web.Mvc;
using AudioStore.Models; // Thay đổi theo namespace Models của bạn

namespace AudioStore.Controllers
{
    public class AccountController : Controller
    {
        private DB_AudioStoreEntities db = new DB_AudioStoreEntities();

        // 1. ĐĂNG KÝ (GET)
        public ActionResult Register()
        {
            return View();
        }

        // 1. ĐĂNG KÝ (POST)
        [HttpPost]
        [ValidateAntiForgeryToken]
        public ActionResult Register(string FullName, string Email, string Password, string ConfirmPassword)
        {
            if (ModelState.IsValid)
            {
                // Kiểm tra Mật khẩu nhập lại
                if (Password != ConfirmPassword)
                {
                    ViewBag.Error = "Mật khẩu xác nhận không khớp!";
                    return View();
                }

                // Kiểm tra trùng lặp Email (hoặc Username)
                var checkEmail = db.Users.FirstOrDefault(u => u.Email == Email || u.Username == Email);
                if (checkEmail != null)
                {
                    ViewBag.Error = "Email này đã được sử dụng!";
                    return View();
                }

                // Tạo User mới
                Users newUser = new Users
                {
                    FullName = FullName,
                    Email = Email,
                    // Vì form Đăng ký của bạn không có ô Username, ta lấy luôn Email làm Username
                    Username = Email,
                    Password = Password, // Lưu ý: Thực tế nên băm (Hash) mật khẩu trước khi lưu
                    RoleID = 2, // 2 là Customer dựa theo DB của bạn
                    CreatedAt = DateTime.Now
                };

                db.Users.Add(newUser);
                db.SaveChanges();

                return RedirectToAction("Login", "Account");
            }
            return View();
        }

        // 2. ĐĂNG NHẬP (GET)
        public ActionResult Login()
        {
            return View();
        }

        // 2. ĐĂNG NHẬP (POST)
        [HttpPost]
        [ValidateAntiForgeryToken]
        public ActionResult Login(string UsernameOrEmail, string Password)
        {
            if (ModelState.IsValid)
            {
                // Tra cứu bằng Username hoặc Email vì giao diện ghi "Email hoặc Tên đăng nhập"
                var user = db.Users.FirstOrDefault(u =>
                    (u.Username == UsernameOrEmail || u.Email == UsernameOrEmail) &&
                    u.Password == Password);

                if (user != null)
                {
                    // Lưu thông tin vào Session
                    Session["User"] = user;
                    Session["FullName"] = user.FullName;
                    Session["RoleID"] = user.RoleID;

                    // Phân quyền chuyển hướng
                    if (user.RoleID == 1) // 1 là Admin
                    {
                        // Đưa Admin nhảy thẳng vào trang lựa chọn Dashboard vừa tạo
                        return RedirectToAction("Index", "Dashboard", new { area = "Admin" }); ;
                    }
                    else // 2 là Customer
                    {
                        return RedirectToAction("Index", "Home");
                    }
                }

                ViewBag.Error = "Tên đăng nhập hoặc mật khẩu không chính xác!";
            }
            return View();
        }

        // 3. ĐĂNG XUẤT
        public ActionResult Logout()
        {
            Session.Clear(); // Xóa toàn bộ Session
            return RedirectToAction("Index", "Home");
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.Mvc;
using AudioStore.Models;

namespace AudioStore.Controllers
{
    public class CartController : Controller
    {
        // Khởi tạo kết nối Database 
        private DB_AudioStoreEntities db = new DB_AudioStoreEntities();

        // ====================================================================
        // HÀM KIỂM TRA ĐĂNG NHẬP (BẢO VỆ GIỎ HÀNG)
        // ====================================================================
        private bool CheckLogin()
        {
            // Kiểm tra xem khách đã đăng nhập chưa (dựa vào Session UserID)
            if (Session["User"] == null)
            {
                return false;
            }
            return true;
        }

        // ====================================================================
        // PHẦN 1: QUẢN LÝ GIỎ HÀNG (DÙNG SESSION)
        // ====================================================================

        // Hàm hỗ trợ: Lấy giỏ hàng từ Session
        private List<CartItem> GetCart()
        {
            var cart = Session["Cart"] as List<CartItem>;
            if (cart == null)
            {
                cart = new List<CartItem>();
                Session["Cart"] = cart;
            }
            return cart;
        }

        // Trang hiển thị Giỏ hàng
        public ActionResult Index()
        {
            // NẾU CHƯA ĐĂNG NHẬP: Trả về một danh sách rỗng để View hiển thị "Giỏ hàng đang trống"
            if (!CheckLogin())
            {
                return View(new List<CartItem>());
            }

            var cart = GetCart();
            return View(cart);
        }

        // Thêm sản phẩm vào giỏ hàng
        public ActionResult AddToCart(int id)
        {
            // BẮT BUỘC ĐĂNG NHẬP: Chưa đăng nhập thì đá về trang Login
            if (!CheckLogin()) return RedirectToAction("Login", "Account");

            var product = db.Products.FirstOrDefault(p => p.ProductID == id && p.IsActive == true);
            if (product != null)
            {
                var cart = GetCart();
                var existingItem = cart.FirstOrDefault(c => c.ProductID == id);

                if (existingItem != null)
                {
                    existingItem.Quantity++; // Nếu đã có thì tăng số lượng
                }
                else
                {
                    // Tách chuỗi ImageUrl bằng dấu '|' để lấy ảnh đầu tiên hiển thị trong giỏ
                    string firstImage = "";
                    if (!string.IsNullOrEmpty(product.ImageUrl))
                    {
                        firstImage = product.ImageUrl.Split('|')[0];
                    }

                    cart.Add(new CartItem
                    {
                        ProductID = product.ProductID,
                        ProductName = product.ProductName,
                        Price = product.Price,
                        Quantity = 1,
                        ImageUrl = firstImage
                    });
                }
            }
            // Trở về trang giỏ hàng sau khi thêm thành công
            return RedirectToAction("Index");
        }

        // Cập nhật số lượng sản phẩm
        [HttpPost]
        public ActionResult UpdateQuantity(int id, int quantity)
        {
            // CHẶN BẰNG URL: Nếu ai đó dùng tool gửi Request thẳng vào đây
            if (!CheckLogin()) return RedirectToAction("Login", "Account");

            var cart = GetCart();
            var item = cart.FirstOrDefault(c => c.ProductID == id);
            if (item != null)
            {
                if (quantity > 0)
                {
                    item.Quantity = quantity;
                }
                else
                {
                    cart.Remove(item); // Nếu số lượng = 0 thì xóa luôn
                }
            }
            return RedirectToAction("Index");
        }

        // Xóa sản phẩm khỏi giỏ
        public ActionResult RemoveFromCart(int id)
        {
            // CHẶN BẰNG URL
            if (!CheckLogin()) return RedirectToAction("Login", "Account");

            var cart = GetCart();
            var item = cart.FirstOrDefault(c => c.ProductID == id);
            if (item != null)
            {
                cart.Remove(item);
            }
            return RedirectToAction("Index");
        }

        // ====================================================================
        // PHẦN 2: LUỒNG THANH TOÁN & BẮT LỖI NGHIÊM NGẶT
        // ====================================================================

        // GET: Hiển thị form thanh toán
        [HttpGet]
        public ActionResult Checkout()
        {

        // POST: Xử lý khi nhấn nút Đặt hàng (Đây là nơi "Bức tường lửa" hoạt động)
        [HttpPost]
        [ValidateAntiForgeryToken] // Chặn tấn công giả mạo Request từ bên ngoài
        public ActionResult Checkout(CheckoutViewModel model)
        {
            if (!CheckLogin()) return RedirectToAction("Login", "Account");

            var cart = GetCart();
            if (cart.Count == 0) return RedirectToAction("Index", "Home");

            // 1. KIỂM TRA DỮ LIỆU: Nếu Giáo viên cố tình sửa F12 hoặc nhập bậy bạ
            if (!ModelState.IsValid)
            {
                // Đá văng về lại trang Checkout kèm theo các dòng chữ đỏ báo lỗi
                return View(model);
            }

            // 2. NẾU VƯỢ QUA ĐƯỢC TẤT CẢ RÀNG BUỘC -> LƯU VÀO DATABASE
            using (var transaction = db.Database.BeginTransaction())
            {
                try
                {
                    // Lấy chính xác ID của người dùng đang đăng nhập từ Session
                    // Lấy ID người dùng từ Session
                    int currentUserId = 0;

                    // Nếu đăng nhập lưu nguyên một Object User
                    if (Session["User"] is AudioStore.Models.Users loggedInUser)
                    {
                        currentUserId = loggedInUser.UserID;
                    }
                    // Hoặc nếu đăng nhập có lưu ID riêng
                    else if (Session["UserID"] != null)
                    {
                        currentUserId = Convert.ToInt32(Session["UserID"]);
                    }
                    // Tính tổng tiền gốc
                    decimal totalAmount = cart.Sum(item => item.TotalPrice);
                    decimal finalAmount = totalAmount;

                    // XỬ LÝ TRỪ TIỀN NẾU CÓ MÃ GIẢM GIÁ
                    if (!string.IsNullOrEmpty(model.DiscountCode))
                    {
                        var discount = db.DiscountCodes.FirstOrDefault(d => d.Code == model.DiscountCode && d.IsActive == true);
                        if (discount != null && (!discount.ExpiryDate.HasValue || discount.ExpiryDate.Value >= DateTime.Now))
                        {
                            decimal discountAmount = (totalAmount * discount.DiscountPercent) / 100;
                            if (discount.MaxDiscountAmount.HasValue && discountAmount > discount.MaxDiscountAmount.Value)
                                discountAmount = discount.MaxDiscountAmount.Value;

                            finalAmount = totalAmount - discountAmount;

                            // Trừ đi 1 lượt sử dụng trong CSDL
                            if (discount.UsageLimit.HasValue)
                            {
                                discount.UsageLimit -= 1;
                            }
                        }
                    }

                    // A. Tạo đơn hàng (Orders) - LƯU GIÁ ĐÃ GIẢM
                    Orders newOrder = new Orders
                    {
                        UserID = currentUserId,
                        OrderDate = DateTime.Now,
                        TotalAmount = totalAmount,
                        FinalAmount = finalAmount, // <- Quan trọng: Cột này sẽ lưu giá đã trừ mã
                        DiscountCode = model.DiscountCode, // <- Quan trọng: Ghi lại mã khách xài
                        ShippingAddress = model.ShippingAddress,
                        PaymentMethod = model.PaymentMethod,
                        Status = "Pending"
                    };

                    db.Orders.Add(newOrder);
                    db.SaveChanges(); // Lưu để EF sinh ra OrderID

                    // B. Tạo chi tiết đơn hàng (OrderDetails)
                    foreach (var item in cart)
                    {
                        OrderDetails detail = new OrderDetails
                        {
                            OrderID = newOrder.OrderID,
                            ProductID = item.ProductID,
                            Quantity = item.Quantity,
                            UnitPrice = item.Price
                        };
                        db.OrderDetails.Add(detail);
                    }
                    db.SaveChanges();

                    transaction.Commit(); // Chốt giao dịch an toàn

                    // Xóa giỏ hàng và chuyển tới trang báo Thành công
                    Session["Cart"] = null;
                    return RedirectToAction("Success");
                }
                catch (Exception ex)
                {
                    transaction.Rollback(); // Nếu có lỗi SQL thì hoàn tác toàn bộ, không lưu rác
                    ModelState.AddModelError("", "Đã xảy ra sự cố khi đặt hàng. Vui lòng thử lại. Lỗi: " + ex.Message);
                    return View(model);
                }
            }
        }

        // ====================================================================
        [HttpPost]
        public JsonResult ApplyDiscount(string code)
        {
            var cart = GetCart();
            if (cart.Count == 0) return Json(new { success = false, message = "Giỏ hàng trống!" });

            decimal totalAmount = cart.Sum(item => item.TotalPrice);

            // Truy vấn CSDL xem mã có tồn tại không
            var discount = db.DiscountCodes.FirstOrDefault(d => d.Code == code && d.IsActive == true);

            if (discount == null)
                return Json(new { success = false, message = "Mã giảm giá không tồn tại hoặc đã bị khóa!" });

            if (discount.ExpiryDate.HasValue && discount.ExpiryDate.Value < DateTime.Now)
                return Json(new { success = false, message = "Mã giảm giá đã hết hạn!" });

            if (discount.MinOrderValue.HasValue && totalAmount < discount.MinOrderValue.Value)
                return Json(new { success = false, message = $"Đơn hàng chưa đạt mức tối thiểu {discount.MinOrderValue.Value:N0}đ để dùng mã này!" });

            if (discount.UsageLimit.HasValue && discount.UsageLimit.Value <= 0)
                return Json(new { success = false, message = "Mã giảm giá này đã hết lượt sử dụng!" });

            // Bắt đầu tính toán tiền giảm
            decimal discountAmount = (totalAmount * discount.DiscountPercent) / 100;

            // Giới hạn số tiền giảm tối đa (Nếu có cấu hình MaxDiscountAmount)
            if (discount.MaxDiscountAmount.HasValue && discountAmount > discount.MaxDiscountAmount.Value)
            {
                discountAmount = discount.MaxDiscountAmount.Value;
            }

            decimal finalAmount = totalAmount - discountAmount;

            return Json(new
            {
                success = true,
                message = "Áp dụng mã giảm giá thành công!",
                discountAmount = discountAmount,
                finalAmount = finalAmount
            });
        }

<script>
    document.addEventListener("DOMContentLoaded", function () {
        var payCOD = document.getElementById("payCOD");
        var payBank = document.getElementById("payBank");
        var qrContainer = document.getElementById("qrCodeContainer");
        var inputDiscount = document.getElementById("inputDiscountCode");
        var btnApply = document.getElementById("btnApplyDiscount");
        var hiddenDiscount = document.getElementById("hiddenDiscountCode");
        var discountRow = document.getElementById("discountRow");
        var discountAmountText = document.getElementById("discountAmountText");
        var finalAmountText = document.getElementById("finalAmountText");
        var discountMessage = document.getElementById("discountMessage");

        // 1. Ẩn / Hiện mã QR khi chuyển đổi phương thức thanh toán
        function toggleQR() {
            if (payBank && payBank.checked) {
                qrContainer.classList.remove("d-none");
            } else if (qrContainer) {
                qrContainer.classList.add("d-none");
            }
        }
        if (payCOD) payCOD.addEventListener("change", toggleQR);
        if (payBank) payBank.addEventListener("change", toggleQR);

        // 2. Chặn ký tự đặc biệt/khoảng trắng khi gõ mã giảm giá (chỉ cho phép chữ và số, tự viết hoa)
        if (inputDiscount) {
            inputDiscount.addEventListener("input", function () {
                this.value = this.value.replace(/[^a-zA-Z0-9]/g, '').toUpperCase();
            });
        }

        // 3. Xử lý khi bấm nút Áp dụng mã giảm giá
        if (btnApply) {
            btnApply.addEventListener("click", function () {
                var code = inputDiscount.value.trim();
                discountMessage.classList.remove("d-none", "text-success", "text-danger");

                if (code === "") {
                    discountMessage.textContent = "Vui lòng nhập mã giảm giá trước khi bấm Áp dụng!";
                    discountMessage.classList.add("text-danger");
                    return;
                }

                btnApply.disabled = true;
                btnApply.textContent = "Đang kiểm tra...";

                fetch('@Url.Action("ApplyDiscount", "Cart")', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body: 'code=' + encodeURIComponent(code)
                })
                .then(function (response) {
                    if (!response.ok) {
                        throw new Error("Không tìm thấy hàm ApplyDiscount trong CartController (Lỗi " + response.status + ")");
                    }
                    return response.json();
                })
                .then(function (res) {
                    if (res.success) {
                        hiddenDiscount.value = code;
                        discountRow.classList.remove("d-none");
                        discountAmountText.textContent = "-" + Number(res.discountAmount).toLocaleString('vi-VN') + " đ";
                        finalAmountText.textContent = Number(res.finalAmount).toLocaleString('vi-VN') + " VNĐ";

                        discountMessage.textContent = "✔ " + res.message;
                        discountMessage.classList.add("text-success");

                        inputDiscount.readOnly = true;
                        btnApply.textContent = "Đã áp dụng";
                        btnApply.disabled = true;
                    } else {
                        discountMessage.textContent = "✖ " + res.message;
                        discountMessage.classList.add("text-danger");
                        btnApply.disabled = false;
                        btnApply.textContent = "Áp dụng";
                    }
                })
                .catch(function (err) {
                    discountMessage.textContent = "✖ Lỗi kết nối: " + err.message;
                    discountMessage.classList.add("text-danger");
                    btnApply.disabled = false;
                    btnApply.textContent = "Áp dụng";
                });
            });
        }
    });
</script>
